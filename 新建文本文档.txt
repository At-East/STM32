#include "./SYSTEM/sys/sys.h"
#include "./SYSTEM/usart/usart.h"
#include "./SYSTEM/delay/delay.h"
#include "./BSP/LED/led.h" //若要控制灯光数量则需修改第三十一行

int main(void)
{
	HAL_Init();
	sys_stm32_clock_init(RCC_PLL_MUL9); //对时钟进行初始化
	delay_init(72); //延时
	usart_init(115200); //串口
	led_init(); //LED
	
	/* 定义引脚数组，放在 while(1) 之前 */
	const uint16_t led_pins[8] = {
    GPIO_PIN_4, GPIO_PIN_5, GPIO_PIN_6, GPIO_PIN_8,
    GPIO_PIN_10, GPIO_PIN_11, GPIO_PIN_12, GPIO_PIN_1
	};

/* 先把所有灯熄灭 */
	for (int i = 0; i < 8; i++) {
    HAL_GPIO_WritePin(GPIOA, led_pins[i], GPIO_PIN_RESET);
	}

		while (1)
		{
			for (int i = 0; i < 8; i++)
			{
        /* 点亮当前引脚 */
        HAL_GPIO_WritePin(GPIOA, led_pins[i], GPIO_PIN_SET);
        delay_ms(500);

        /* 熄灭当前引脚，再处理下一个 */
        HAL_GPIO_WritePin(GPIOA, led_pins[i], GPIO_PIN_RESET);
			}

	
	/*while(1)

}
}
